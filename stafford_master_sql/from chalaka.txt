CREATE TABLE shipments (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    country_of_origin VARCHAR(10) NOT NULL,
    buyer_id VARCHAR(255) DEFAULT NULL,
    shipment_id VARCHAR(255) NOT NULL,
    shipment_id_mode VARCHAR(10) NOT NULL,
    purchase_order_id BIGINT UNSIGNED DEFAULT NULL,
    shipment_type VARCHAR(20) NOT NULL,
    factory_ids JSON NOT NULL,
    created_at TIMESTAMP NULL DEFAULT NULL,
    updated_at TIMESTAMP NULL DEFAULT NULL,
    UNIQUE (shipment_id)
);

CREATE TABLE shipment_order_factories (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    shipment_id BIGINT UNSIGNED NOT NULL,
    purchase_order_id INT NOT NULL,
    factory_id INT NOT NULL,
    CONSTRAINT shipment_po_factory_unique UNIQUE (shipment_id, purchase_order_id, factory_id),
    CONSTRAINT shipment_order_factories_shipment_id_foreign FOREIGN KEY (shipment_id) REFERENCES shipments(id) ON DELETE CASCADE
);

-- 1. Create the temporary JSON column
ALTER TABLE shipments ADD COLUMN purchase_order_ids_copy JSON DEFAULT NULL;

-- 2. Migrate and aggregate data into the new JSON column
WITH AggregatedPOs AS (
    SELECT id, JSON_ARRAYAGG(po_id) as po_ids
    FROM (
        -- Get POs directly from shipments table
        SELECT id, purchase_order_id AS po_id 
        FROM shipments 
        WHERE purchase_order_id IS NOT NULL
        
        UNION
        
        -- Get related POs from the pivot table
        SELECT shipment_id AS id, purchase_order_id AS po_id 
        FROM shipment_order_factories 
        WHERE purchase_order_id IS NOT NULL
    ) AS unique_pos
    GROUP BY id
)
UPDATE shipments s
LEFT JOIN AggregatedPOs a ON s.id = a.id
SET s.purchase_order_ids_copy = a.po_ids;

-- 3. Drop the old BIGINT column
ALTER TABLE shipments DROP COLUMN purchase_order_id;

-- 4. Rename the temporary JSON column to the final column name
ALTER TABLE shipments CHANGE purchase_order_ids_copy purchase_order_id JSON NULL;


-- 1. Create the temporary BIGINT column
ALTER TABLE shipments ADD COLUMN purchase_order_ids_copy BIGINT UNSIGNED DEFAULT NULL;

-- 2. Migrate data back, keeping the ID only if there is exactly 1 PO in the array
UPDATE shipments 
SET purchase_order_ids_copy = JSON_UNQUOTE(JSON_EXTRACT(purchase_order_id, '$[0]'))
WHERE JSON_LENGTH(purchase_order_id) = 1;

-- 3. Drop the JSON column
ALTER TABLE shipments DROP COLUMN purchase_order_id;

-- 4. Rename the temporary BIGINT column to the final column name
ALTER TABLE shipments CHANGE purchase_order_ids_copy purchase_order_id BIGINT UNSIGNED NULL;
